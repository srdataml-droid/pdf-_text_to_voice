cmake_minimum_required(VERSION 3.22)
project(anchor_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(NOT ANCHOR_LLAMA_DIR)
  set(ANCHOR_LLAMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp")
endif()
add_subdirectory("${ANCHOR_LLAMA_DIR}" llama-build EXCLUDE_FROM_ALL)
add_library(anchor_llama SHARED anchor_llama.cpp)
target_link_libraries(anchor_llama PRIVATE llama android log)
target_compile_options(anchor_llama PRIVATE -Wall -Wextra -Werror=return-type)
target_link_options(anchor_llama PRIVATE "-Wl,-z,max-page-size=16384")
